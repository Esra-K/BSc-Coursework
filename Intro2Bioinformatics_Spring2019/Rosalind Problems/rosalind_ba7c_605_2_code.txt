import numpy as np
from collections import defaultdict

def limb(j, mat):

    i = (j+1) % len(mat)
    k = (j+2) % len(mat)

    minn = (mat[i, j] + mat[j, k] - mat[i, k]) / 2

    for i in range(len(mat)):
        for k in range(i+1,len(mat)):
            if i!=j and k!=j:
                lenn = (mat[i, j] + mat[j, k] - mat[i, k]) / 2
                minn = min(minn,lenn)

    return minn

def additive_triple(d, l):
    for i in range(l):
        for k in range(i + 1, l):
            if d[i, k] == d[i, l] + d[l, k]:
                return (i, k)
    return

def add_phy(node_count, mat):
    if node_count == 2:
        return
    limbo = limb(node_count - 1, mat[:node_count, :node_count])
    for j in range(node_count - 1):
        mat[node_count - 1, j] -= limbo
        mat[j, node_count - 1] -= limbo
    (i, k) = additive_triple(mat, node_count - 1)
    element = mat[i, node_count - 1]
    add_phy(node_count - 1, mat)
    add_leaf(node_count - 1, i, k, element, limbo)
    return
