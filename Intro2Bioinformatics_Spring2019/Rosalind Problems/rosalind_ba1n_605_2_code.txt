def d_neighbors(str, d, alphabet):
    if d == 0:
        return [str]
    recursive = d_neighbors(str[1:], d - 1, alphabet)
    result = [first + restOfNeighbor for first in alphabet if first != str[0] for restOfNeighbor in recursive]

    if (d < len(str)):
        recursive = d_neighbors(str[1:], d, alphabet)
        result = result + [str[0] + r3 for r3 in recursive]
    return result


str = input()
d = int(input())
result = d_neighbors(str, d, alphabet=['A', 'C', 'G', 'T'])
for _ in result:
    print(_)
print(str)
