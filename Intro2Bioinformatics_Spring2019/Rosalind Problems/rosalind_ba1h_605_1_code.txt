substring = input()
string = input()
d = int(input())
indices = []
for j in range(len(string) - len(substring) + 1):
    sub1 = string[j:j + len(substring)]
    hamburger = 0
    for i in range(len(substring)):
        if substring[i] != sub1[i]:
            hamburger += 1
    if hamburger <= d:
        indices.append(j)
for index in range(len(indices)):
    if index < len(indices) - 1:
        print(indices[index], end=" ")
    else:
        print(indices[index], end="")
